def escolher_raca():
    while True:
        raca = input(
            "\nDigite qual raça deseja ser:"
            "\n- Anão"
            "\n- Draconato"
            "\n- Elfo"
            "\n- Gnomo"
            "\n- Halfling"
            "\n- Humano"
            "\n- Meio-Elfo"
            "\n- Meio-Orc"
            "\n- Tiefling"
            "\n\nR: "
        ).strip().lower()

        if raca not in RACAS:
            print("Raça inválida!")
            continue

        subtipos = RACAS[raca]

        if subtipos:
            while True:
                print(f"\nSub-raças disponíveis para {raca.title()}:")
                for subtipo in subtipos:
                    print(f"- {subtipo}")

                sub = input("Digite a sub-raça: ").strip().lower()

                if sub in subtipos:
                    raca_completa = f"{raca} ({sub})"
                    break

                print("Sub-raça inválida!")

        elif raca == "humano":
            while True:
                tipo = input("Deseja ser Humano comum ou variante? ").strip().lower()

                if tipo in ("comum", "variante"):
                    raca_completa = f"humano ({tipo})"
                    break

                print("Tipo de humano inválido!")

        else:
            raca_completa = raca

        confirma = input(
            f"Tem certeza que deseja ser {raca_completa.title()}? (s/n): "
        ).strip().lower()

        if confirma == "s":
            return raca_completa


def pedir_nivel():
    while True:
        try:
            nivel = int(input("Digite o seu nível de personagem de 1 a 20: "))

            if 1 <= nivel <= 20:
                return nivel

            print("Nível inválido! Digite um valor entre 1 e 20.")

        except ValueError:
            print("Entrada inválida! Digite um número inteiro.")


def calcular_proficiencia(nivel):
    if nivel <= 4:
        return 2
    if nivel <= 8:
        return 3
    if nivel <= 12:
        return 4
    if nivel <= 16:
        return 5
    return 6

def rolar_atributo():
    dados = [random.randint(1, 6) for _ in range(4)]
    dados.remove(min(dados))
    return sum(dados)


def distribuir_aleatorio():
    atributos = {}
    rolagens = sorted([rolar_atributo() for _ in range(6)], reverse=True)

    print(f"\nSeus 6 valores gerados foram: {rolagens}")

    pendentes = ATRIBUTOS_BASE.copy()

    for valor in rolagens:
        while True:
            escolha = input(
                f"Escolha qual atributo receberá o valor {valor} "
                f"({', '.join(pendentes)}): "
            ).strip().lower()

            if escolha in pendentes:
                atributos[escolha] = valor
                pendentes.remove(escolha)
                break

            print("Atributo inválido ou já preenchido!")

    return atributos


def distribuir_padrao():
    valores_padrao = [15, 14, 13, 12, 10, 8]

    print(f"\nOs valores padrão são: {valores_padrao}")

    while True:
        entrada = input(
            "Digite os 6 atributos separados por vírgula na ordem do maior "
            "para o menor valor\n"
            "(Ex: força, destreza, constituição, inteligência, sabedoria, carisma):\n"
            "R: "
        ).strip().lower()

        lista = [a.strip() for a in entrada.split(",")]

        if len(lista) == 6 and set(lista) == set(ATRIBUTOS_BASE):
            return dict(zip(lista, valores_padrao))

        print(
            "Entrada inválida! Certifique-se de listar exatamente os 6 "
            "atributos sem repetir."
        )


def distribuir_compra():
    tabela_custos = {
        8: 0,
        9: 1,
        10: 2,
        11: 3,
        12: 4,
        13: 5,
        14: 7,
        15: 9,
    }

    atributos = {atributo: 8 for atributo in ATRIBUTOS_BASE}
    pontos_restantes = 27

    while True:
        print("\n" + "-" * 40)
        print(f"Pontos restantes: {pontos_restantes}")

        for nome, valor in atributos.items():
            print(f"  {nome.title()}: {valor}")

        if pontos_restantes == 0:
            confirma = input(
                "\nVocê gastou todos os pontos! Deseja finalizar? (s/n): "
            ).strip().lower()

            if confirma == "s":
                return atributos

        escolha = input(
            "\nQual atributo deseja alterar (ou 'fim' para terminar)? "
        ).strip().lower()

        if escolha == "fim":
            return atributos

        if escolha not in atributos:
            print("Atributo inválido!")
            continue

        try:
            novo_valor = int(
                input(f"Digite o novo valor para {escolha.title()} (8 a 15): ")
            )
        except ValueError:
            print("Por favor, digite um número inteiro válido.")
            continue

        if novo_valor not in tabela_custos:
            print("Valor inválido! Deve ser entre 8 e 15.")
            continue

        custo_atual = tabela_custos[atributos[escolha]]
        custo_novo = tabela_custos[novo_valor]
        diferenca = custo_novo - custo_atual

        if diferenca > pontos_restantes:
            print(
                f"Pontos insuficientes! Faltam "
                f"{diferenca - pontos_restantes} ponto(s)."
            )
            continue

        atributos[escolha] = novo_valor
        pontos_restantes -= diferenca


def escolher_atributo_especial(atributos, quantidade, escolhidos=None):
    if escolhidos is None:
        escolhidos = []

    for _ in range(quantidade):
        while True:
            escolha = input(
                f"Escolha um atributo para receber +1 "
                f"(opções: {', '.join(ATRIBUTOS_BASE)}): "
            ).strip().lower()

            if escolha not in ATRIBUTOS_BASE:
                print("Atributo inválido!")
                continue

            if escolha in escolhidos:
                print("Você precisa escolher um atributo diferente!")
                continue

            atributos[escolha] += 1
            escolhidos.append(escolha)
            break


def bonus_raca(atributos, raca):
    if raca in SUB_RACAS:
        for atributo, valor in SUB_RACAS[raca].items():
            atributos[atributo] += valor

    elif raca == "humano (comum)":
        for atributo in ATRIBUTOS_BASE:
            atributos[atributo] += 1

    elif raca == "humano (variante)":
        escolher_atributo_especial(atributos, 2)

    elif raca == "meio-elfo":
        atributos["carisma"] += 2
        escolher_atributo_especial(atributos, 2)


def distribuir_atributos():
    while True:
        forma = input(
            "\nDigite qual forma deseja distribuir seus pontos "
            "(Aleatorio / Padrão / Compra): "
        ).strip().lower()

        if forma in ("aleatorio", "aleatório"):
            atributos = distribuir_aleatorio()

        elif forma in ("padrão", "padrao"):
            atributos = distribuir_padrao()

        elif forma == "compra":
            atributos = distribuir_compra()

        else:
            print("Opção inválida!")
            continue

        confirma = input(
            "\nTem certeza da sua escolha de distribuição? (s/n): "
        ).strip().lower()

        if confirma == "s":
            return atributos


def calcular_modificadores(atributos):
    return {
        atributo: (valor - 10) // 2
        for atributo, valor in atributos.items()
    }

def escolher_armadura_e_arma_marcial():
    itens = []

    while True:
        escolha_armadura = input(
            "Deseja (1) uma cota de malha, ou "
            "(2) uma armadura de couro? Digite 1 ou 2: "
        ).strip()

        if escolha_armadura == "1":
            itens.append("cota de malha")
            break

        if escolha_armadura == "2":
            itens.append("armadura de couro")
            itens.append("arco longo")
            break

        print("Escolha inválida!")

    while True:
        escolha_arma = input(
            "Deseja (1) uma arma marcial e um escudo, ou "
            "(2) duas armas marciais? Digite 1 ou 2: "
        ).strip()

        if escolha_arma == "1":
            itens.append(input("Digite a arma marcial escolhida: ").strip())
            itens.append("escudo")
            break

        if escolha_arma == "2":
            itens.append(input("Digite a primeira arma marcial escolhida: ").strip())
            itens.append(input("Digite a segunda arma marcial escolhida: ").strip())
            break

        print("Escolha inválida!")

    return itens

def escolher_equipamentos(classe):
    if classe == "guerreiro":
        return equipamentos_guerreiro()

    if classe == "paladino":
        return equipamentos_paladino()

    itens = []

    for pergunta, fixo in EQUIPAMENTOS_POR_CLASSE[classe]:
        if fixo is not None:
            itens.append(fixo)
        else:
            itens.append(input(pergunta).strip())

    return itens


def confirmar_equipamentos(classe):
    while True:
        print(
            "Escolha seus itens iniciais. Sempre que aparecer "
            "'(escreva)' na pergunta, digite o item desejado."
        )

        itens = escolher_equipamentos(classe)

        confirma = input(
            "Esses são os equipamentos iniciais que deseja? (s/n) "
        ).strip().lower()

        if confirma == "s":
            return itens


def calcular_pv(dado_vida, nivel, modificador_constituicao):
    pv_nivel_1 = dado_vida + modificador_constituicao

    pv_por_nivel = [
        max(1, random.randint(1, dado_vida) + modificador_constituicao)
        for _ in range(nivel - 1)
    ]

    return pv_nivel_1 + sum(pv_por_nivel)


def montar_ficha():
    print("=====| CRIAR FICHA DUNGEONS & DRAGONS |=====")

    raca = escolher_raca()
    classe, nivel = escolher_classe()

    dado_vida_base = CLASSES_DADO_VIDA[classe]
    dado_vida_texto = f"{nivel}d{dado_vida_base}"
    proficiencia = calcular_proficiencia(nivel)

    atributos = distribuir_atributos()

    bonus_raca(atributos, raca)

    modificadores = calcular_modificadores(atributos)

    pv = calcular_pv(
        dado_vida_base,
        nivel,
        modificadores["constituição"],
    )

    descricao = pedir_descricao()
    equipamentos = confirmar_equipamentos(classe)

    print("\n===== FICHA CRIADA =====")
    print(f"Raça: {raca.title()}")
    print(f"Classe: {classe.title()} - Nível {nivel}")
    print(f"Dado de vida: {dado_vida_texto}")
    print(f"Bônus de proficiência: +{proficiencia}")
    print(f"Pontos de vida: {pv}")

    print("\nAtributos (Modificadores):")

    for atributo in ATRIBUTOS_BASE:
        valor = atributos[atributo]
        mod = modificadores[atributo]
        sinal = "+" if mod >= 0 else ""
        print(f"  {atributo.title()}: {valor} ({sinal}{mod})")

    print(f"\nDescrição:\n{descricao}")

    print("\nEquipamentos iniciais:")

    for item in equipamentos:
        print(f"  - {item}")


if __name__ == "__main__":
    montar_ficha()
